const { readFileSync } = require('fs');
const path = './input.txt'; // Path to file containing OCR results

// Function to split large blocks of texts into individual segments
function segmentArticles(textBlock){
    const regexPattern = /\n\s*\n/g;
    return textBlock.split(regexPattern);
}

// Function to identify headlines and corresponding contents
function parseArticle(articleSegment){
    const lines = articleSegment.trim().split('\n');

    // Heuristic approach to determine whether a line could potentially serve as a headline
    function isHeadline(line){
        return /^[^a-z]*$/i.test(line); // Headlines often consist entirely of uppercase letters
    }

    let headline = '';
    let contentStartIndex = 0;

    for(let index = 0; index < lines.length; ++index){
        if(isHeadline(lines[index])){
            headline = lines[index].trim();
            contentStartIndex = index+1;
            break;
  	}
    }

    const contentLines = lines.slice(contentStartIndex);

    // Combine multiline content back together
    const content = contentLines.join(" ").replace(/\s+/g,' ').trim();

    return {
        headline: headline || "NA",
        content: content || "NA"
    };
}


// Main Execution Block
try{
	const rawData = readFileSync(path, 'utf-8'); // Read entire file
    
    // Splitting the full text into smaller manageable sections representing different articles
    const segmentedBlocks = segmentArticles(rawData);

    // Parsing each identified segment/article separately
    const articlesArray = [];
    segmentedBlocks.forEach(segment => {
      	if(segment && segment.trim()){
          	let parsedArticle = parseArticle(segment);
            	articlesArray.push(parsedArticle);
    	}	
    });

    console.log(JSON.stringify({data: articlesArray},null,2)); // Print final structured json representation
}catch(err){
	console.error(`Error occurred: ${err.message}`);
}
