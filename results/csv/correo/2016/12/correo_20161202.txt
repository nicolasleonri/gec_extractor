def parse_newspaper_content(raw_text):
    # Splitting input text into individual lines
    lines = raw_text.split('\n')
    
    # Initialize variables to store current state
    articles = []
    current_article = {"headline": None, "content": ""}
    
    def add_current_article():
        """Helper method to finalize and save the currently processed article."""
        nonlocal current_article
        if current_article["headline"] != "NA":
            articles.append(current_article.copy())
        
    for line in lines:
        stripped_line = line.strip()
        
        # Check for headlines which typically appear at the start of an article
        if len(stripped_line) >= 40 and any(char.isupper() for char in stripped_line[:15]):
            # Finalize previous article before starting a new one
            add_current_article()

            # Set up new article entry
            current_article = {"headline": stripped_line, "content": ""}

        elif current_article["headline"]:
            # Append content to the ongoing article
            if current_article["content"].endswith(" ") or not current_article["content"]:
                current_article["content"] += stripped_line + "\n"

    # Add last collected article after loop ends
    add_current_article()

    return [{"headline": item.get('headline'), "content": item['content'].strip()} for item in articles]
