# Splitting the cleaned up text into segments potentially representing separate articles
segments = input_text.strip().split('---')

for seg in segments:
    # Attempting to identify possible headlines which often appear bold or capitalized
    parts = seg.split('\n')
    headline_candidate = None
    
    for part in parts:
        if '**' in part:  # Assuming titles might have been marked using markdown style bold syntax
            headline_candidate = part.replace('*', '').strip()
            
    if headline_candidate:
        start_index = parts.index(headline_candidate) + 1
        content_lines = []
        
        while start_index < len(parts):  
            line = parts[start_index].replace('*','').strip()  # Removing any remaining asterisks used for emphasis
            
            if line == '' or line.isupper():  # Skipping blank lines or additional uppercase headings
                break
                
            content_lines.append(line)
            start_index += 1
            
        full_content = ' '.join(content_lines)

            "headline": headline_candidate,
            "content": full_content
        })
